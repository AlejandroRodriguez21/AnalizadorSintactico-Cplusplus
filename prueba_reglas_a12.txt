
class ReglasOperaciones {
    int x = 8;
    int y = 3;
    float factor = 1.5;
    bool activo = verdadero;

    void ejecutar() {
        int resta = x - y;
        int modulo = x % y;
        float mezcla = x + factor;
        bool comparacion = x >= y;
        comparacion = x <= y;
        comparacion = x != y;
        comparacion = x == y;
        comparacion = activo || falso;
    }
}
