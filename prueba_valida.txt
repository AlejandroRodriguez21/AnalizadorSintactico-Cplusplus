class Principal {
    void iniciar() {
        escribir("Inicio del programa");
    }

    void mostrar(int valor) {
        escribir(valor);
    }

    int sumar(int a, int b) {
        retornar a + b;
    }

    int obtenerValor() {
        retornar 100;
    }

    int calcular(int x, int y) {
        int resultado = sumar(x,y);
        si (resultado > 100) {
            resultado = resultado * 2;
        } sino {
            resultado = resultado + 10;
        }
        retornar resultado;
    }

    void proceso() {
        iniciar();
        mostrar(10);
        int total = calcular(20,30);
        escribir(total);
    }
}
