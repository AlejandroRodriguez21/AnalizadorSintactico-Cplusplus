class Principal {
    void proceso() {
        si edad >= 18 {
            escribir("Mayor");
        }

        mientras (contador < 10 {
            contador = contador + 1;
        }

        si (activo) {
            escribir("Activo");
        

        sino {
            escribir("Inactivo");
        }

        para (int i=0; i < 10;) {
            escribir(i);
        }

        segun (opcion) {
            caso:
                escribir("Opcion");
                romper;
        }
    }
}
